rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /{document=**} {
      allow read, write: if false;
    }

    match /players/{userId} {
      allow read: if request.auth != null; 
      allow write: if request.auth.uid == userId; 
    }

    match /players/{userId}/notes/{noteId} {
      allow read, write: if request.auth.uid == userId;
    }

    match /players/{userId}/passwords/{passwordId} {
      allow read, write: if request.auth.uid == userId;
    }

    match /players/{userId}/friendRequests/{requestId} {
      allow read, write: if request.auth.uid == userId || request.auth.uid == requestId;
    }

    match /players/{userId}/friends/{friendId} {
      allow read: if request.auth != null; 
      allow write: if request.auth.uid == userId || request.auth.uid == friendId;
    }

    match /players/{userId}/notifications/{notificationId} {
      allow read, write: if request.auth.uid == userId;
      allow create: if request.auth.uid != userId && 
        request.resource.data.keys().hasAll(['type', 'fromUserId', 'message', 'createdAt']);
    }

    // Updated messages rule for group chats
    match /messages/{messageId} {
      allow read, write: if request.auth != null &&
        request.auth.uid in resource.data.participants;
      allow create: if request.auth != null &&
        request.auth.uid in request.resource.data.participants &&
        request.resource.data.participants.size() >= 2 &&
        request.resource.data.participants.size() <= 50;
    }

    // New group chats collection
    match /groupChats/{groupId} {
      allow read: if request.auth != null &&
        request.auth.uid in resource.data.members;
      allow create: if request.auth != null &&
        request.auth.uid in request.resource.data.members &&
        request.auth.uid == request.resource.data.createdBy;
      allow update: if request.auth != null &&
        request.auth.uid in resource.data.members &&
        (request.auth.uid == resource.data.createdBy || 
         request.auth.uid in resource.data.admins);
    }

    match /presence/{userId} {
      allow read: if request.auth != null;
      allow write: if request.auth.uid == userId;
    }

    match /typing/{typingId} {
      allow read, write: if request.auth != null && 
        (typingId.matches('.*_' + request.auth.uid + '$') || 
         typingId.matches('^' + request.auth.uid + '_.*'));
    }

    match /players/{userId}/sessions/{sessionId} {
      allow read, write: if request.auth.uid == userId;
    }

    match /players/{userId}/loginHistory/{historyId} {
      allow read, write: if request.auth.uid == userId;
    }

    match /players/{userId}/securityEvents/{eventId} {
      allow read, write: if request.auth.uid == userId;
    }

    match /players/{userId}/backupCodes/{codeId} {
      allow read, write: if request.auth.uid == userId;
    }

    match /emailVerifications/{userId} {
      allow read, write: if request.auth.uid == userId;
    }
  }
}
